function dkIsFiniteNumber(value) {
    return typeof value === "number" &&
        value === value &&
        value < (1 / 0) &&
        value > (-1 / 0);
}

function dkIsInteger(value) {
    return dkIsFiniteNumber(value) &&
        Math.floor(value) === value;
}

function dkIsArray(value) {
    return !!value &&
        typeof value === "object" &&
        typeof value.length === "number" &&
        typeof value.push === "function";
}

function dkThrow(message) {
    throw message;
}



function dkRequireInteger(value, minimum, message) {
    if (!dkIsInteger(value) || value < minimum) {
        dkThrow(message);
    }
}

function dkParseJSON(text) {
    var position = 0;

    function skipWhitespace() {
        while (position < text.length &&
            (text.charAt(position) === " " ||
                text.charAt(position) === "\n" ||
                text.charAt(position) === "\r" ||
                text.charAt(position) === "\t")) {
            position++;
        }
    }

    function parseString() {
        var result = "";
        position++;
        while (position < text.length) {
            var character = text.charAt(position++);
            if (character === '"') {
                return result;
            }
            if (character !== "\\") {
                result += character;
                continue;
            }
            if (position >= text.length) {
                dkThrow("DK: incomplete string escape.");
            }
            var escaped = text.charAt(position++);
            if (escaped === '"' || escaped === "\\" || escaped === "/") {
                result += escaped;
            } else if (escaped === "b") {
                result += "\b";
            } else if (escaped === "f") {
                result += "\f";
            } else if (escaped === "n") {
                result += "\n";
            } else if (escaped === "r") {
                result += "\r";
            } else if (escaped === "t") {
                result += "\t";
            } else if (escaped === "u") {
                var code = parseInt(text.substr(position, 4), 16);
                if (isNaN(code)) {
                    dkThrow("DK: invalid Unicode escape.");
                }
                result += String.fromCharCode(code);
                position += 4;
            } else {
                dkThrow("DK: invalid string escape.");
            }
        }
        dkThrow("DK: unterminated string.");
    }

    function parseValue() {
        skipWhitespace();
        var character = text.charAt(position);
        if (character === '"') {
            return parseString();
        }
        if (character === "[") {
            position++;
            var array = [];
            skipWhitespace();
            if (text.charAt(position) === "]") {
                position++;
                return array;
            }
            while (position < text.length) {
                array.push(parseValue());
                skipWhitespace();
                character = text.charAt(position++);
                if (character === "]") {
                    return array;
                }
                if (character !== ",") {
                    dkThrow("DK: invalid array separator.");
                }
            }
            dkThrow("DK: unterminated array.");
        }
        if (character === "{") {
            position++;
            var object = {};
            skipWhitespace();
            if (text.charAt(position) === "}") {
                position++;
                return object;
            }
            while (position < text.length) {
                skipWhitespace();
                if (text.charAt(position) !== '"') {
                    dkThrow("DK: object key must be a string.");
                }
                var key = parseString();
                skipWhitespace();
                if (text.charAt(position++) !== ":") {
                    dkThrow("DK: invalid object separator.");
                }
                object[key] = parseValue();
                skipWhitespace();
                character = text.charAt(position++);
                if (character === "}") {
                    return object;
                }
                if (character !== ",") {
                    dkThrow("DK: invalid object separator.");
                }
            }
            dkThrow("DK: unterminated object.");
        }
        if (text.substr(position, 4) === "true") {
            position += 4;
            return true;
        }
        if (text.substr(position, 5) === "false") {
            position += 5;
            return false;
        }
        if (text.substr(position, 4) === "null") {
            position += 4;
            return null;
        }
        var numberMatch = text.slice(position).match(
            /^-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?/
        );
        if (numberMatch) {
            position += numberMatch[0].length;
            return Number(numberMatch[0]);
        }
        dkThrow("DK: invalid JSON value.");
    }

    var value = parseValue();
    skipWhitespace();
    if (position !== text.length) {
        dkThrow("DK: unexpected data after JSON value.");
    }
    return value;
}

function dkPaethPredictor(left, up, upperLeft) {
    var estimate = left + up - upperLeft;
    var leftDistance = Math.abs(estimate - left);
    var upDistance = Math.abs(estimate - up);
    var upperLeftDistance = Math.abs(estimate - upperLeft);
    if (leftDistance <= upDistance && leftDistance <= upperLeftDistance) {
        return left;
    }
    if (upDistance <= upperLeftDistance) {
        return up;
    }
    return upperLeft;
}

function dkUnfilterPNGScanlines(scanlines, width, height) {

function dkChooseFrame(dat, frame) {
    var count = dkFrameCount(dat);
    var frameNumber = 0;
    var frameIndex = 0;
    var hasDelays = false;
    var totalTicks = 0;
    var elapsed;
    var duration;
    var i;
    if (frame !== undefined) {
        frameNumber = Math.floor(frame);
    }
    if (!dkIsFiniteNumber(frameNumber) || frameNumber < 0) {
        frameNumber = 0;
    }
    frameIndex = frameNumber % count;
    if (dkIsArray(dat.t)) {
        hasDelays = true;
    } else if (dkIsFiniteNumber(dat.t) && dat.t > 0) {
        hasDelays = true;
    }
    if (hasDelays) {
        for (i = 0; i < count; i++) {
            totalTicks += dkFrameDurationAt(dat, i);
        }
        elapsed = frameNumber % totalTicks;
        for (i = 0; i < count; i++) {
            duration = dkFrameDurationAt(dat, i);
            if (elapsed < duration) {
                frameIndex = i;
                break;
            }
            elapsed -= duration;
        }
    }
    return frameIndex;
}

        packed[i] = (deltaBytes[i] + previousBytes[i]) & 255;
    }
    return packed;
}

function dkIndexBytesFromPacked(packedBytes, paletteBytes) {
    var indexes;
    var i;
    if (!paletteBytes) {
        return packedBytes;
    }
    indexes = dkNewByteArray(packedBytes.length - paletteBytes);
    for (i = 0; i < indexes.length; i++) {
        indexes[i] = packedBytes[paletteBytes + i];
    }
    return indexes;
}

function dkPreviousIndexBytes(dat, frameIndex) {
    if (dat._indexBytes && dat._indexBytesFrame === frameIndex - 1) {
        return dat._indexBytes;
    }
    var previous = null;
    var packed;
    var i;
    for (i = 0; i < frameIndex; i++) {
        packed = dkSliceFrame(dat, i);
        if (i === 0) {
            previous = dkIndexBytesFromPacked(packed, dat.c * 4);
        } else {
            previous = dkApplyFrameDelta(packed, previous);
        }
    }
    return previous;
}

/* Decode and render the selected lossless image frame. */
function drawLosslessImage(startX, startY, pixelSize, dat, frame) {
    var frameIndex;
    var rgba;
    if (!dat || dat.k !== "DKL") {
        dkThrow("DK: unsupported image format.");
    }
    if (typeof createGraphics !== "function" || typeof image !== "function") {
        dkThrow("DK: rendering functions unavailable.");
    }
    if (!dkIsFiniteNumber(pixelSize) || pixelSize <= 0) {
        dkThrow("DK: invalid pixel size.");
    }
    dkRequireInteger(dat.w, 1, "DK: invalid image width.");
    dkRequireInteger(dat.h, 1, "DK: invalid image height.");
    if (!dkIsInteger(dat.w * dat.h * 4) || dat.w * dat.h * 4 > 100000000) {
        dkThrow("DK: image dimensions are too large.");
    }
    frameIndex = dkChooseFrame(dat, frame);
    rgba = dkUnfilterPNGScanlines(dkSliceFrame(dat, frameIndex), dat.w, dat.h);
    dkDrawCached(dat, frameIndex, rgba, startX, startY, pixelSize);
}

function drawColorImage(startX, startY, pixelSize, dat, frame) {
    var frameIndex;
    var packed;
    var rgba;
    var pixelCount;
    if (!dat || dat.k !== "DKC5") {
        dkThrow("DK: unsupported image format.");
    }
    if (typeof createGraphics !== "function" || typeof image !== "function") {
        dkThrow("DK: rendering functions unavailable.");
    }
    if (!dkIsFiniteNumber(pixelSize) || pixelSize <= 0) {
        dkThrow("DK: invalid pixel size.");
    }
    dkRequireInteger(dat.w, 1, "DK: invalid image width.");
    dkRequireInteger(dat.h, 1, "DK: invalid image height.");
    pixelCount = dat.w * dat.h;
    if (!dkIsInteger(pixelCount * 4) || pixelCount * 4 > 100000000) {
        dkThrow("DK: image dimensions are too large.");
    }
    frameIndex = dkChooseFrame(dat, frame);
    if (frameIndex > 0 && !dat._palette) {
        dkDecodePalette(dkSliceFrame(dat, 0), dat, pixelCount, 0);
    }
    packed = dkSliceFrame(dat, frameIndex);
    if (frameIndex > 0) {
        packed = dkApplyFrameDelta(packed, dkPreviousIndexBytes(dat, frameIndex));
        dat._indexBytes = packed;
    } else {
        dat._indexBytes = dkIndexBytesFromPacked(packed, dat.c * 4);
    }
    dat._indexBytesFrame = frameIndex;
    rgba = dkDecodePalette(packed, dat, pixelCount, frameIndex);
    dkDrawCached(dat, frameIndex, rgba, startX, startY, pixelSize);
}

function drawReadablePixelArt(startX, startY, pixelSize, dat, frame) {
    if (!dat || dat.k !== "DKP2" || !dkIsArray(dat.frames) ||
        dat.frames.length === 0) {
        dkThrow("DKP2: no pixel-art frames found.");
    }
    if (!dkIsFiniteNumber(pixelSize) || pixelSize <= 0) {
        dkThrow("DKP2: invalid pixel size.");
    }
    if (typeof fill !== "function" ||
        typeof noStroke !== "function" ||
        typeof rect !== "function") {
        dkThrow("DKP2: rectangle drawing functions unavailable.");
    }
    var frameNumber = frame === undefined ? 0 : Math.floor(frame);
    if (!dkIsFiniteNumber(frameNumber) || frameNumber < 0) {
        frameNumber = 0;
    }
    var frameIndex = frameNumber % dat.frames.length;
    var hasDelays = false;
    var totalTicks = 0;
    var i;
    var duration;
    for (i = 0; i < dat.frames.length; i++) {
        if (dat.frames[i] && dkIsFiniteNumber(dat.frames[i].delay) &&
            dat.frames[i].delay > 0) {
            hasDelays = true;
            break;
        }
    }
    if (hasDelays) {
        for (i = 0; i < dat.frames.length; i++) {
            totalTicks += dkFrameDuration(dat.frames[i]);
        }
        var elapsed = frameNumber % totalTicks;
        for (i = 0; i < dat.frames.length; i++) {
            duration = dkFrameDuration(dat.frames[i]);
            if (elapsed < duration) {
                frameIndex = i;
                break;
            }
            elapsed -= duration;
        }
    }
    var pixelFrame = dat.frames[frameIndex];
    if (!pixelFrame || !dkIsArray(pixelFrame.canvasData) ||
        pixelFrame.canvasData.length !== dat.height ||
        !pixelFrame.colorMap ||
        typeof pixelFrame.colorMap !== "object") {
        dkThrow("DKP2: invalid frame data.");
    }
    var preserveStyle = typeof pushStyle === "function" &&
        typeof popStyle === "function";
    if (preserveStyle) {
        pushStyle();
    }
    try {
        noStroke();
        if (typeof rectMode === "function" &&
            typeof CORNER !== "undefined") {
            rectMode(CORNER);
        }
        var y;
        var row;
        var x;
        var symbol;
        var runEnd;
        var mappedColor;
        var left;
        var top;
        var right;
        var bottom;
        for (y = 0; y < dat.height; y++) {
            row = pixelFrame.canvasData[y];
            if (typeof row !== "string" || row.length !== dat.width) {
                dkThrow("DKP2: row " + y + " has an invalid width.");
            }
            x = 0;
            while (x < dat.width) {
                symbol = row.charAt(x);
                runEnd = x + 1;
                while (runEnd < dat.width && row.charAt(runEnd) === symbol) {
                    runEnd++;
                }
                if (symbol !== "-") {
                    mappedColor = pixelFrame.colorMap[symbol];
                    if (mappedColor === undefined ||
                        mappedColor === null) {
                        dkThrow("DKP2: unmapped pixel symbol.");
                    }
                    fill(mappedColor);
                    left = Math.round(startX + x * pixelSize);
                    top = Math.round(startY + y * pixelSize);
                    right = Math.round(startX + runEnd * pixelSize);
                    bottom = Math.round(startY + (y + 1) * pixelSize);
                    rect(left, top, right - left, bottom - top);
                }
                x = runEnd;
            }
        }
    } finally {
        if (preserveStyle) {
            popStyle();
        }
    }
}

function drawHugePixelArt(
    startX,
    startY,
    pixelSize,
    canvasData,
    unusedColorMap,
    frame
) {
    if (canvasData && canvasData.k === "DKP2") {
        drawReadablePixelArt(
            startX,
            startY,
            pixelSize,
            canvasData,
            frame
        );
        return;
    }
    if (canvasData && canvasData.k === "DKC5") {
        drawColorImage(
            startX,
            startY,
            pixelSize,
            canvasData,
            frame
        );
        return;
    }
    if (!canvasData || canvasData.k !== "DKL") {
        dkThrow(
            "DK renderer received unsupported data."
        );
    }
    drawLosslessImage(
        startX,
        startY,
        pixelSize,
        canvasData,
        frame
    );
}
